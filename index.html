// Leaderboard System (Only real usernames, limited to Top 5) let playerStats = JSON.parse(localStorage.getItem('smashLeaderboard')) || []; function renderLeaderboard() { const list = document.getElementById('leaderboardList'); list.innerHTML = ''; // Filter out blank or default "Untitled" names, sort by most played, and grab top 5 const validPlayers = playerStats .filter(p => p.name && p.name.trim() !== '' && p.name.toLowerCase() !== 'untitled') .sort((a, b) => b.games - a.games) .slice(0, 5); if (validPlayers.length === 0) { list.innerHTML = `
No players registered yet. Set your username and create a match!
`; return; } const badgeIcons = ['🥇', '🥈', '🥉', '4️⃣', '5️⃣']; validPlayers.forEach((player, index) => { const row = document.createElement('div'); row.className = 'smash-card rounded-2xl p-4 flex justify-between items-center text-white'; row.innerHTML = `
${badgeIcons[index] || '🛡️'}
${player.name}
🎮 ${player.games} Matches Played
`; list.appendChild(row); }); } function registerOrUpdatePlayer(username, addGames = 1) { const cleanName = username ? username.trim() : ""; // Ignore blank or "Untitled" usernames if (!cleanName || cleanName.toLowerCase() === 'untitled') return; let found = playerStats.find(p => p.name.toLowerCase() === cleanName.toLowerCase()); if (found) { found.games += addGames; } else { playerStats.push({ name: cleanName, games: addGames }); } localStorage.setItem('smashLeaderboard', JSON.stringify(playerStats)); renderLeaderboard(); }
